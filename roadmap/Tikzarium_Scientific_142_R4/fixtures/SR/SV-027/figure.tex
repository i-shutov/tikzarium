  mark=none,samples=50,smooth}, axis lines=middle,
  axis x line=bottom, axis y line=none, ytick=\empty,
  xtick=\empty,
  extra x ticks={0, \S},
  extra x tick labels={$\omega_A$, $\omega_0$},
  xlabel=$\omega$,

  enlargelimits=false] \addplot[blue,domain=-8:8,name path=B,thick] {4*gauss(x,0,\Bs)};

    \path[name path=xaxis]
      (0,0) -- (\pgfkeysvalueof{/pgfplots/xmax},0); \addplot[white!50!green] fill between[of=xaxis and S];
